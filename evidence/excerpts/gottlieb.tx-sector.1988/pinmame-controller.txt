304: static void rockwell10939_ld(Rockwell10939State* state, UINT8 data) {
305:    if (state->isCtrlDataWord == 0 && data == 0x01) {
306:       state->isCtrlDataWord = 1; // Next data is a control word
307:    }
308:    else if (state->isCtrlDataWord == 1 && data != 0x01) {
309:       state->isCtrlDataWord = 0;
310:       if (data == 0x05)
311:          state->nDigitCycle = 16;
312:       else if (data == 0x06)
313:          state->nDigitCycle = 32;
314:       else if (data == 0x07)
315:          state->nDigitCycle = 64;
316:       // 0x08 0x09 0x0A are cusrsor control mode (not implemented)
317:       else if (data == 0x0e)
318:          state->nHalted = 0; // Start Display Refresh Cycle (start display strobing, also start second slave display driver)
319:       else if (data >= 0x40 && data <= 0x7F)
320:          state->dutyCycle = data & 0x7F; // Duty cycle (brightness, relative to nDigitCycle - 1)
321:       else if (data >= 0x80 && data <= 0x9F)
322:          state->nDigits = data == 0x80 ? 32 : (data & 0x1F); // Digit Counter Register: number of digits to drive
323:       else if (data >= 0xC0 && data <= 0xD3)
324:          state->nWriteCol = data & 0x1F; // Next write column
325:       //if (data != 0xc0)
326:       //   printf("%8.5f Driver %02d: Control data word 1: %02x\n", timer_get_time(), state->coreSegPos, data);
327:    }
328:    else if (state->nDigits) {
329:       //printf("%8.5f Driver %02d: %02x\n", timer_get_time(), state->coreSegPos + state->nWriteCol, data);
330:       const UINT16 seg16 = core_ascii2seg16[data & 0x7f];
331:       coreGlobals.segments[state->coreSegPos + state->nWriteCol].w = seg16;
332:       const float brightness = (state->dutyCycle == 0) ? 1.f : (float)(state->dutyCycle + 1) / (float)state->nDigitCycle;
333:       for (int i = 0, pos = state->corePWMSegPos + state->nWriteCol * 16; i < 16; i++, pos++)
334:         coreGlobals.physicOutputState[pos].value = (seg16 & (1 << i)) ? brightness : 0.f;
335:       state->display[state->nWriteCol] = data;
336:       state->nWriteCol = (state->nWriteCol + 1) % state->nDigits;
337:    }
338: }
